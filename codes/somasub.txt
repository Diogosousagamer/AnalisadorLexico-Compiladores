program somasub;
var x,y,resultado,opcao: integer;

/* inicio do 
programa */
begin 
	println("Numero X:");
	read(x);

	println("Numero Y:");
	read(y);

	println("Deseja realizar: 1 - Soma; 2 - Subtracao");
	read(opcao);

	if (opcao = 1) then
		resultado := x + y;
	else
		resultado := x - y;

	writeln(resultado);
end
