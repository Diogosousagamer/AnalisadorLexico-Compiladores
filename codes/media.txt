program media;
var nota1, nota2, nota3, media: real;

begin
	println("Digite sua nota na primeira unidade: ");
	read(nota1);
	
	println("Digite sua nota na segunda unidade: ");
	read(nota2);
	
	println("Digite sua nota na terceira unidade: ");
	read(nota3);

	media := (nota1 + nota2 + nota3) / 3;	

	println("Media total: " + media);

	if media >= 7.0 then
		println("Aprovado");
	else
		println("Reprovado");
end.
